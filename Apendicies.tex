\chapter{Proof that a Symmetric Energy Distribution fails to cancel in the Product-Sum}
\chaptermark{Product-Sum Proof}
\label{ch:symm_sample}
	In the ansatz reflected in equation (\ref{eq:simple_sym_eec}), a symmetric set with a distribution roughly approximating a Gaussian was utilized, however, we can go beyond that to properly show this result is the case in a set drawn from a generalized Gaussian distribution
	\newline
	\begin{proof}{}
		We begin with a Gaussian distribution $P(x)$ with amplitude $A$ and sample points from it such that we construct a "Gaussian sample set" with the following properties 
		\newline
		\begin{definition}[Gaussian Sample Set]
		\label{def:GSS}
		Take a set S that is constructed of points randomly sampled such that  
		\begin{equation}
			S \approx P(x)=  \frac{A}{\sqrt{2 \pi \sigma}} e^\frac{-x^2}{\sigma}
		\end{equation}
		Which gives the following properties
		\begin{enumerate}
		\item{ $ S= \left\{ x | \bar{x} = \delta_{\bar{x}} \right\}$} 
		\item{ $ \frac{1}{N} \sum_{x\in S} x^2 = \left( \sigma + \delta_\sigma \right)^2$}
		\item{ $ \left| \delta_{\bar{x}} \right|, \left| \delta_\sigma \right| << 1 $}
		\item{ $\lim_{N\rightarrow \infty} \left| \delta_{\bar{x}} \right|, \left| \delta_\sigma \right| = 0 $}
		\end{enumerate}
		
		\end{definition}
		
		Then, the Product-Sum is calculated by following the standard algorithm 
		\newline
		\begin{algo}[Product-Sum]
		\begin{equation}
			\begin{split}
				\forall{x} \in {S} & s.t. len(S) > 1\\
				& S = S / x  \\
				& PS = PS + ( x * y ) \hspace{1.5 cm} \forall y \in S \\	
			\end{split}	
		\end{equation}
		\textit{Product-sum algorithm for the sample set S. Initially the PS variable is set at 0}
		\end{algo}
		Using the properties of S noted in definition \ref{def:GSS}, one notes that for any sampled value x, there should be approximately $n= \floor{\frac{A}{\sqrt{2 \pi (\sigma+ \delta_\sigma)}} e^{\frac{\left( x - \delta_{\bar{x}} \right)^2}{\sigma+\delta_\sigma}}}$ of the same value. 
		In fact, by including these small corrections and an additional small correction $\left|\delta_{n, x}\right| << 1$  we fix this number to the Gaussian curve within some fluctuations. 
	With this in mind, one breaks the contributions to the Product-sum into two categories, the $y=x$ and $y \neq x$ case. 
	For the first case, the contribution is simply 
	\begin{equation}
		PS_{y=x} = x^2 \left( n(x) - 1 \right)! 
	\end{equation}
	and in the second case 
	\begin{equation}
		PS_{y\neq x} = x y n(x) n(y)
	\end{equation}
	Thus the product-sum becomes
	\begin{equation}
		PS = \sum_{x\in S}\left[  x^2 \left(n(x) - 1\right)! + \sum_{y > x, y\in S}x y n(x)n(y) \right] 
	\end{equation}
	With a sufficiently high number of samples, we can shift to the continuum limit and write
	\begin{equation}
		\widetilde{PS} = \int_{-x_0}^{x_0} \dd x \hspace{0.2 cm} \left[ x^2 \Gamma(\left(n(x)\right)) + \int_{x}^{x_{0}} \dd y \frac{xy}{2 \pi \left( \sigma + \delta \sigma \right)} e^{\frac{- \left( x - \delta_{\bar{x}} \right)^2 - \left( y - \delta_{\bar{x}} \right)^2}{2 \left( \sigma + \delta \sigma\right)}}   \right]
	\end{equation}
	where $x_0$ is a limit set by finding the point at which there is only 1 measurable point
		\begin{equation}
			n(x_0) = 1 \Rightarrow x_0 =\delta_{\bar{x}}+ \sqrt{2 (\sigma+\delta \sigma) \ln A - (\sigma+\delta \sigma) \ln \left[ 2 \pi \left( \sigma + \delta \sigma \right) \right]}
		\end{equation}
	Taking the part of this integral where of $x \neq y$, one finds
		\begin{equation}
			\begin{split}
				\widetilde{PS}_{x\neq y} &= \int_{-x_0}^{x_0} \dd x \frac{x}{2 \pi \left( \sigma + \delta \sigma \right)} e^{\frac{- \left( x - \delta_{\bar{x}} \right)^2}{2 \left( \sigma + \delta \sigma \right) }} \int_{x_0 - \delta_{\bar{y}} }^{x_0} \dd w \left( w + \delta_{\bar{y}} \right) e^{\frac{- w^2}{2 \left( \sigma + \delta \sigma \right)} }\\
				&= \int_{-\infty}^{\infty} \dd x \frac{x}{2 \pi \left( \sigma + \delta \sigma \right)} e^{\frac{- \left( x - \delta_{\bar{x}} \right)^2}{2 \left( \sigma + \delta \sigma \right) }} \left[ \left( \sigma + \delta \sigma \right) e^{\frac{- \left( x - \delta_{\bar{y}} \right)^2}{2 \left( \sigma + \delta \sigma \right) }} + \delta_{\bar{y}} \int_{x-\delta_{\bar{y}}}^\infty \dd w e^{\frac{- w^2}{2 \left( \sigma + \delta \sigma \right) }} \right]\\
				&= \int_{-\infty}^{\infty} \dd x \frac{x}{2 \pi} e^{\frac{- \left( x - \frac{\delta_{\bar{x}} + \delta_{\bar{y}}}{2} \right)}{\sigma + \delta \sigma} + \frac{\left( \delta_{\bar{x}} - \delta_{\bar{y}}\right)^2}{4(\sigma+ \delta \sigma)}} + \int_{-\infty}^{\infty} \dd x \frac{x}{2 \pi \left( \sigma + \delta \sigma \right)} e^{\frac{- \left( x - \delta_{\bar{x}} \right)^2}{2 \left( \sigma + \delta \sigma \right) }}\delta_{\bar{y}} \int_{x-\delta_{\bar{y}}}^\infty \dd w e^{\frac{- w^2}{2 \left( \sigma + \delta \sigma \right) }} \\
				&= \frac{1}{2\pi} \frac{\delta_{\bar{x}} + \delta_{\bar{y}}}{2} \sqrt{\pi \left( \sigma + \delta \sigma \right) } e^{\frac{\left( \delta_{\bar{x}} - \delta_{\bar{y}}\right)^2}{4(\sigma+ \delta \sigma)}} +  \int_{-\infty}^{\infty} \dd x \frac{x}{2 \pi \left( \sigma + \delta \sigma \right)} e^{\frac{- \left( x - \delta_{\bar{x}} \right)^2}{2 \left( \sigma + \delta \sigma \right) }} \delta_{\bar{y}} \int_{x-\delta_{\bar{y}}}^\infty \dd w e^{\frac{- w^2}{2 \left( \sigma + \delta \sigma \right) }} \\
				& = 0   +  \int_{-\infty}^{\infty} \dd x \frac{x}{2 \pi \left( \sigma + \delta \sigma \right)} e^{\frac{- \left( x - \delta_{\bar{x}} \right)^2}{2 \left( \sigma + \delta \sigma \right) }} \times 0 \\ 
				&= 0 
			\end{split}
		\end{equation}
	Where the last step notes that $\lim_{N \rightarrow \infty} \delta_{\bar{x}}, \delta_{\bar{y}} = 0 $ by property 4 of the definition \ref{def:GSS}, but is not necessarily 0 for a discrete sample set 

	Thus the product sum goes simply to 
	\begin{equation}
		\widetilde{PS} = \int_{- x_0}^{x_0} \dd x \hspace{0.5 cm} x^2 \Gamma\left( \frac{A}{\sqrt{2 \pi (\sigma + \delta \sigma)}} e^{\frac{-\left(x- \delta_{\bar{x}}\right)^2}{2(\sigma+ \delta\sigma)}} \right) 
		\label{eq:gamma_int}
	\end{equation}
	Which is always positive, given that the integrand is always positive as is shown in fig. \ref{fig:gamma_integrand}, thus the product sum is non-zero even in the continuum limit, thus for a Gaussian distribution centered around 0, the sum of two point products is not guaranteed to be positive and thus the symmetry of the distribution can not be exploited to cancel out the noise profile
	\end{proof}
